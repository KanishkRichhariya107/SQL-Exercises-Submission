CREATE TABLE email_subscribers (
    id SERIAL PRIMARY KEY,
    email TEXT,
    city TEXT,
    phone TEXT
);

QUERY 1

SELECT distinct(city) FROM email_subscribers


QUERY 2

SELECT city, count(*) AS responded_people
FROM email_subscribers
GROUP BY city;

QUERY 3

SELECT city, count(*) AS responded_people
FROM email_subscribers
GROUP BY city
ORDER BY responded_people DESC
LIMIT 1;

QUERY 4

SELECT distinct(split_part(email, '@', 2)) AS email_domains
FROM email_subscribers;

QUERY 5

SELECT distinct(split_part(email, '@', 2)) AS email_domains, COUNT(*) as count
FROM email_subscribers
GROUP BY split_part(email, '@',2)
ORDER BY count DESC
LIMIT 1;
